def add_address():

def calculate_distance(point1, point2):
    return sqrt(
        (point1[0] - point2[0]) ** 2 +
        (point1[1] - point2[1]) ** 2
    )

def nearest_neighbor_route(start_address, addresses):

    geolocator = Nominatim(user_agent="courier_app", timeout=5)

    try:
        start_location = geolocator.geocode(start_address)

        if not start_location:
            return addresses

        current_point = (
            start_location.latitude,
            start_location.longitude
        )

    except Exception:
        return addresses

    remaining = addresses.copy()
    ordered_route = []

    while remaining:

        nearest_address = None
        nearest_distance = float("inf")

        for address in remaining:

            try:
                location = geolocator.geocode(address)

                if not location:
                    continue

                point = (
                    location.latitude,
                    location.longitude
                )

                distance = calculate_distance(current_point, point)

                if distance < nearest_distance:
                    nearest_distance = distance
                    nearest_address = address

            except Exception:
                continue

        if nearest_address:
            ordered_route.append(nearest_address)
            remaining.remove(nearest_address)

            try:
                location = geolocator.geocode(nearest_address)

                if location:
                    current_point = (
                        location.latitude,
                        location.longitude
                    )
            except Exception:
                pass

        else:
            break

    for address in remaining:
        ordered_route.append(address)

    return ordered_route

def generate_route():

    global addresses

    if not start_address:
        set_status("Najpierw podaj adres startowy")
        return

    if not addresses:
        set_status("Brak adresów do wygenerowania trasy")
        return

    listbox.delete(0, tk.END)

    ordered_addresses = nearest_neighbor_route(start_address, addresses)

    addresses = ordered_addresses

    for index, addr in enumerate(addresses, start=1):
        listbox.insert(tk.END, f"{index}. {addr}")

    set_status("Trasa została wygenerowana")

def refresh_listbox():
    listbox.delete(0, tk.END)

    for index, addr in enumerate(addresses, start=1):
        listbox.insert(tk.END, f"{index}. {addr}")

def delete_address():

    global start_address

    selected = listbox.curselection()

    if selected:

        index = selected[0]

        removed_address = addresses[index]

        if index == 0:
            start_address = removed_address
            set_status("Punkt startowy został zaktualizowany")

        addresses.pop(index)

        cursor.execute("""
        DELETE FROM addresses
        WHERE postal_code || ' ' || city || ', ' || street || ' ' || building_number = ?
        """, (removed_address,))

        conn.commit()

        refresh_listbox()
        show_package_count()
        set_status("Adres został usunięty")
        update_route_summary()
    else:
        set_status("Zaznacz adres do usunięcia")

def show_map():
    geolocator = Nominatim(user_agent="courier_app")

    mapa = folium.Map(
        location=[52.2297, 21.0122],
        zoom_start=11
    )

    route_points = []

    if start_address:
        try:
            start_location = geolocator.geocode(start_address)

            if start_location:
                start_point = [start_location.latitude, start_location.longitude]
                route_points.append(start_point)

                folium.Marker(
                    start_point,
                    popup=f"START: {start_address}",
                    tooltip="START",
                    icon=folium.DivIcon(
                        html="""
                        <div style="
                            background-color: #1E3A5F;
                            color: white;
                            border: 2px solid #FFD447;
                            border-radius: 8px;
                            padding: 5px;
                            font-weight: bold;">
                            START
                        </div>
                        """
                    )
                ).add_to(mapa)

        except Exception:
            set_status("Nie udało się znaleźć adresu startowego")

    for index, address in enumerate(addresses, start=1):
        try:
            location = geolocator.geocode(address)

            if location:
                point = [location.latitude, location.longitude]
                route_points.append(point)

                folium.Marker(
                    point,
                    popup=f"{index}. {address}",
                    tooltip=f"Punkt {index}: {address}",
                    icon=folium.DivIcon(
                        html=f"""
                        <div style="
                            background-color: #FFD447;
                            color: black;
                            border: 2px solid black;
                            border-radius: 50%;
                            width: 28px;
                            height: 28px;
                            text-align: center;
                            line-height: 28px;
                            font-weight: bold;
                            font-size: 14px;">
                            {index}
                        </div>
                        """
                    )
                ).add_to(mapa)

        except Exception:
            pass

    if len(route_points) >= 2:
        folium.PolyLine(
            route_points,
            weight=5,
            opacity=0.8
        ).add_to(mapa)

    mapa.save("mapa_trasy.html")
    webbrowser.open("mapa_trasy.html")
    set_status("Mapa została wygenerowana")

def new_route():

    answer = messagebox.askyesno(
        "Nowa trasa",
        "Czy na pewno usunąć wszystkie zapisane adresy?"
    )

        listbox.delete(0, tk.END)

        cursor.execute("DELETE FROM addresses")

        conn.commit()

        show_package_count()
        set_status("Rozpoczęto nową trasę")
        update_route_summary()
