import tkinter as tk
import sqlite3
import folium 
import webbrowser


from geopy.geocoders import Nominatim


addresses = []


conn = sqlite3.connect("courier.db")
cursor = conn.cursor()

cursor.execute("""
CREATE TABLE IF NOT EXISTS addresses (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    city TEXT,
    postal_code TEXT,
    street TEXT,
    building_number TEXT
)
""")

conn.commit()

def show_package_count():
    count_label.config(text=f"Liczba paczek: {len(addresses)}")


def add_address():

    city = city_entry.get()
    postal = postal_entry.get()
    street = street_entry.get()
    number = number_entry.get()

    if city and postal and street and number:

        full_address = f"{postal} {city}, {street} {number}"

        addresses.append(full_address)

        listbox.insert(tk.END, full_address)

        # zapis do bazy danych
        cursor.execute("""
        INSERT INTO addresses (city, postal_code, street, building_number)
        VALUES (?, ?, ?, ?)
        """, (city, postal, street, number))

        conn.commit()

        # czyszczenie pól
        city_entry.delete(0, tk.END)
        postal_entry.delete(0, tk.END)
        street_entry.delete(0, tk.END)
        number_entry.delete(0, tk.END)

        show_package_count()


def generate_route():

    listbox.delete(0, tk.END)

    sorted_addresses = sorted(addresses)

    for index, addr in enumerate(sorted_addresses, start=1):

        listbox.insert(tk.END, f"{index}. {addr}")


def delete_address():

    selected = listbox.curselection()

    if selected:

        index = selected[0]

        listbox.delete(index)

        addresses.pop(index)

        show_package_count()


def show_map():
    geolocator = Nominatim(user_agent="courier_app")

    mapa = folium.Map(
        location=[52.2297, 21.0122],
        zoom_start=11
    )

    route_points = []

    for index, address in enumerate(addresses, start=1):
        try:
            location = geolocator.geocode(address)

            if location:
                point = [location.latitude, location.longitude]
                route_points.append(point)

                folium.Marker(
                    point,
                    popup=f"{index}. {address}",
                    tooltip=f"Punkt {index}: {address}",
                    icon=folium.DivIcon(
                        html=f"""
                        <div style="
                            background-color: #FFD447;
                            color: black;
                            border: 2px solid black;
                            border-radius: 50%;
                            width: 28px;
                            height: 28px;
                            text-align: center;
                            line-height: 28px;
                            font-weight: bold;
                            font-size: 14px;">
                            {index}
                        </div>
                        """
                    )
                ).add_to(mapa)

        except Exception:
            pass

    if len(route_points) >= 2:
        folium.PolyLine(
            route_points,
            weight=5,
            opacity=0.8
        ).add_to(mapa)

    mapa.save("mapa_trasy.html")
    webbrowser.open("mapa_trasy.html")
